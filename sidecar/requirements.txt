# Jura Trace — Python ML Sidecar Dependencies (runtime / production)
#
# This file covers ONLY runtime dependencies — what the frozen PyInstaller
# binary and CI installs need to run. Heavy ML training dependencies
# (torch, open_clip) live in requirements-build.txt, which is never installed
# in CI or production.
#
# Pin strategy: all production packages use exact versions (==) to prevent
# silent upgrades from introducing breaking changes or security regressions.
# Use `pip-compile requirements.txt --generate-hashes --output-file requirements.lock`
# to regenerate requirements.lock after any version bump here.
# CI should install from requirements.lock (with hashes) and run pip-audit
# against it to detect known CVEs before deployment.
#
# Requirements split:
#   requirements.txt      — this file; runtime baseline installed in CI + production
#   requirements-build.txt — adds open-clip-torch + torch for ONNX export + training
#   requirements-ci.txt   — fully-pinned mirror of the runtime baseline for CI

# HEIC/HEIF support — required for iPhone photos on Linux production builds.
# macOS often works without this via system ImageIO but Linux PyInstaller
# bundles will silently fail to open .heic/.heif files. The sidecar calls
# pillow_heif.register_heif_opener() at startup; if the import fails we
# log CRITICAL and continue (HEIC verification will fail, other formats OK).
pillow-heif==1.6.0

# AVIF support — added rc.31 (2026-06-09) after a known-fake test image in AVIF
# format passed the trust algorithm at 77% because six detectors returned HTTP
# 400 'cannot identify image file' and four returned score=0 ('clean'),
# inflating the composite. pillow-heif 1.6.0 does not include AVIF; this plugin
# adds an AVIF reader to PIL via a separate AvifImagePlugin auto-registered on
# import. macOS sips/Preview have native AVIF but PyInstaller bundles ship a
# self-contained Python interpreter, so AVIF must come from a pip package.
pillow-avif-plugin==1.5.5

# Copy-move detection clustering

# Deepfake detection (texture, frequency, JPEG analysis)

# ONNX runtime — explicit pin required: production CLIP detection runs via
# onnxruntime (JTV-143, 3 May 2026). Without this pin onnxruntime would only
# appear as a transitive resolution from open-clip-torch, which has been
# moved to requirements-build.txt. Omitting this pin causes onnxruntime to
# silently disappear from requirements.lock, breaking CLIP at runtime.
onnxruntime==1.24.4

# regex — PCRE-compatible pattern library required by app/services/clip_tokenizer.py
# (imported as `import regex as re`). Previously a transitive dep via open_clip_torch;
# moving open_clip to requirements-build.txt would silently drop regex from the
# runtime lock without this explicit pin.
regex==2025.10.23
